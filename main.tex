% =====================================================================
%  5. TITRES DES PARTIES ET SECTIONS
% =====================================================================
% Parties (chapitres) : "Partie I : TITRE", 16pt gras, aligné à gauche
\titleformat{\chapter}[block]
  {\normalfont\bfseries}
  {\fontsize{16}{20}\selectfont\MakeUppercase{Partie \thechapter}~:}
  {0.4em}
  {\raggedright\normalfont\fontsize{16}{20}\selectfont\bfseries\MakeUppercase}
\titleformat{name=\chapter,numberless}[block]
  {\normalfont\bfseries}
  {}
  {0pt}
  {\raggedright\normalfont\fontsize{16}{20}\selectfont\bfseries\MakeUppercase}
\titlespacing*{\chapter}{0pt}{0pt}{54pt}
\titlespacing*{name=\chapter,numberless}{0pt}{0pt}{54pt}

\renewcommand{\thechapter}{\Roman{chapter}}

% Sections : "I.1 Titre", 14pt gras
\titleformat{\section}
  {\normalfont\fontsize{14}{17}\selectfont\bfseries}
  {\thesection}
  {0.6em}
  {}
\titlespacing*{\section}{0pt}{12pt}{12pt}

% Sous-sections : "I.1.1 Titre", 12pt gras
\titleformat{\subsection}
  {\normalfont\fontsize{12}{15}\selectfont\bfseries}
  {\thesubsection}
  {0.6em}
  {}
\titlespacing*{\subsection}{0pt}{6pt}{6pt}

% Subdivisions : "I.1.1.1 Titre", 12pt gras
\titleformat{\subsubsection}
  {\normalfont\fontsize{12}{15}\selectfont\bfseries}
  {\thesubsubsection}
  {0.6em}
  {}
\titlespacing*{\subsubsection}{0pt}{6pt}{6pt}
